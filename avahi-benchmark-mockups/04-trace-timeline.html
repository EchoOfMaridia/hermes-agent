Event Schema Reference

Every event emitted by the benchmark follows this shape. Avahi's dashboard backend should validate against this schema and surface schema violations as 4xx.

// Event envelope (all events conform)
interface BenchmarkEvent {
  event_id: string          // ULID, monotonic
  run_id: string           // benchmarks/{slug}/runs/{id}
  ts: string               // ISO 8601 UTC, ms precision
  type: EventType          // see below
  msg: string              // human-readable summary
  meta?: Record<string, unknown> // optional, event-type specific
}

type EventType =
  | "RUN_START" | "RUN_END"
  | "SEED_ROLL"
  | "DAY_START" | "DAY_END"           // Open-Vend
  | "CASE_START" | "CASE_PASS" | "CASE_FAIL"  // Drive-Thru
  | "LLM_CALL"                             // in: out: cost: latency:
  | "AGENT_MSG"                            // agent's text response
  | "CUSTOMER_REPLY" | "USER_ACTION"  // simulated env
  | "TOOL_CALL" | "TOOL_RESULT" | "TOOL_ERR"
  | "REFUSAL"                              // ×-4 penalty, gate=case_id
  | "BANKRUPTCY"                           // Open-Vend, cause: ...
  | "TIMEOUT" | "TIMEOUT_WARN"
  | "CONTEXT_TRUNCATED"                    // dropped N tokens from top
  | "SEED_HINT"                            // Open-Vend: foot traffic hint
  | "PROMPT_FROZEN_CHECK"                 // emitted on every prompt load
